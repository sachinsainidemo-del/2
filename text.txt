        window.onload = function() {
            const canvas = document.getElementById('smoke-canvas');
            const ctx = canvas.getContext('2d');
            const wrapper = document.getElementById('intro-wrapper');
            const envelope = document.getElementById('password-screen');
            canvas.width = window.innerWidth; canvas.height = window.innerHeight;
            const particles = [];
            class Particle {
                constructor() {
                    this.x = canvas.width / 2; this.y = canvas.height / 2;
                    this.vx = (Math.random() - 0.5) * 20; this.vy = (Math.random() - 0.5) * 20;
                    this.alpha = 1; this.size = Math.random() * 40 + 20;
                    this.color = Math.random() > 0.5 ? '#fff' : '#ffe4e6';
                }
                draw() {
                    ctx.globalAlpha = this.alpha; ctx.fillStyle = this.color;
                    ctx.beginPath(); ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
                    ctx.fill(); ctx.filter = 'blur(10px)';
                }
                update() { this.x += this.vx; this.y += this.vy; this.alpha -= 0.01; this.size += 0.5; }
            }
            setTimeout(() => {
                wrapper.classList.add('open');
                for(let i=0; i<80; i++) particles.push(new Particle());
                function animate() {
                    ctx.clearRect(0,0, canvas.width, canvas.height);
                    particles.forEach((p, index) => { p.update(); p.draw(); if(p.alpha <= 0) particles.splice(index, 1); });
                    if(particles.length > 0) requestAnimationFrame(animate);
                }
                animate();
                setTimeout(() => {
                    envelope.classList.add('visible');
                    canvas.style.opacity = '0';
                    document.body.style.overflow = 'auto';
                }, 800);
            }, 1000);
        };

        const revealObserver = new IntersectionObserver((entries) => {
            entries.forEach(entry => { if (entry.isIntersecting) entry.target.classList.add('active'); });
        }, { threshold: 0.15 });

        const passField = document.getElementById('password-field');
        passField.addEventListener('input', (e) => {
            if (e.target.value === '1803') { unlockArchive(); }
            else if (e.target.value.length === 4) {
                document.getElementById('error-text').style.display = 'block';
                e.target.value = '';
            }
        });

        function unlockArchive() {
            document.getElementById('password-screen').style.opacity = '0';
            setTimeout(() => {
                document.getElementById('password-screen').style.display = 'none';
                const main = document.getElementById('main-content');
                main.style.display = 'block';
                setTimeout(() => {
                    main.style.opacity = '1';
                    document.querySelectorAll('.reveal').forEach(el => revealObserver.observe(el));
                    setInterval(createPetal, 700);
                }, 100);
            }, 1000);
        }

        function createPetal() {
            const petal = document.createElement('div');
            petal.className = 'petal';
            petal.style.setProperty('--left', Math.random() * 100 + 'vw');
            const size = Math.random() * 15 + 10;
            petal.style.width = size + 'px'; petal.style.height = size + 'px';
            document.getElementById('petals-container').appendChild(petal);
            setTimeout(() => petal.remove(), 10000);
        }

            document.getElementById('overlay-dark').style.display = 'block';

            document.getElementById('overlay-dark').style.display = 'none';
